import requests

url = "https://YOUR-GRADIO-LINK.gradio.live/api/predict_with_key"
payload = {
    "data": [
        "What is the integral of x² sin(x)?",
        "am-xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx"
    ]
}

res = requests.post(url, json=payload)
print(res.json()["data"][0])
const res = await fetch(
  "https://YOUR-GRADIO-LINK.gradio.live/api/predict_with_key",
  {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      data: [
        "Solve x² - 5x + 6 = 0",
        "am-xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx"
      ]
    })
  }
);
const result = await res.json();
console.log(result.data[0]);
